+    async function loadCommunityDriveCatalogOverview(options) {
+      options = options || {};
+      try {
+        const url = communityCatalogDriveUrl() + (options.forceRefresh ? '?refresh=1' : '');
+        const res = await fetch(url, { method: 'GET', headers: { Accept: 'application/json' } });
+        if (!res.ok) {
+          console.warn('[catalog] Drive overview HTTP', res.status);
+          return null;
+        }
+        const data = await res.json();
+        if (data && data.grades && typeof data.grades === 'object') {
+          appState.communityDriveCatalogOverview = data;
+          return data;
+        }
+      } catch (err) {
+        console.warn('[catalog] Drive overview failed:', err && err.message ? err.message : err);
+      }

+    }
