===== JS =====
    function placeCommunitySummarizerBgHost() {
      var host = document.getElementById('community-summarizer-bg-host');
      if (!host) return;
      var target = document.getElementById('community-summarizer-bg-float-mount');
      if (target && host.parentNode !== target) {
        target.appendChild(host);
      }
    }

    function syncCommunitySummarizerBgIndicator() {
      var badge = document.getElementById('community-summarizer-bg-indicator');
      if (!badge) return;
      placeCommunitySummarizerBgHost();
      var host = document.getElementById('community-summarizer-bg-host');
      var modalOpen = isCommunitySummarizerModalOpen();
      var summarizing = Boolean(appState.isSummarizing || appState.communitySummarizerLoading);
      var show = Boolean(appState.communitySummarizerBgPending) && !modalOpen
        && (summarizing || appState.communitySummarizerResult);
      var ready = show && !summarizing && Boolean(appState.communitySummarizerResult);
      var statusEl = document.getElementById('community-summarizer-bg-status');
      var iconEl = badge.querySelector('.community-summarizer-bg-icon');
      var tip = ready ? t('community_summarizer_bg_title_ready') : t('community_summarizer_bg_title_working');
      badge.classList.toggle('hidden', !show);
      badge.classList.toggle('is-ready', ready);
      badge.setAttribute('aria-hidden', show ? 'false' : 'true');
      badge.setAttribute('title', tip);
      if (host) host.classList.toggle('is-visible', show);
      if (statusEl) {
        statusEl.setAttribute('data-i18n', ready ? 'community_summarizer_bg_ready' : 'community_summarizer_bg_working');
        statusEl.textContent = ready ? t('community_summarizer_bg_ready') : t('community_summarizer_bg_working');
      }
      if (iconEl) {
        iconEl.className = ready
          ? 'fa-solid fa-circle-check community-summarizer-bg-icon'
          : 'fa-solid fa-clock fa-spin community-summarizer-bg-icon';
      }
